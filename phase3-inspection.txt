}) {

        <Link
          className="btn small"
          href={href}
        >
          View gift
        </Link>
      </div>
    </article>
  );
}

'use client';

import {
  useRef,
  useState,
} from 'react';

import Link from 'next/link';

import {
  getPersonalisationProduct,
  TextLayout,
} from '../lib/personalisationConfig';

import {
  products as shopProducts,
} from '../lib/products';

import {
  makeCode,
} from '../lib/storage';

import ConfiguredProductPreview from './ConfiguredProductPreview';

type Props = {
  slug: string;
};

export default function ProductPersonaliser({
  slug,
}: Props) {
  const product =
    getPersonalisationProduct(slug);

  const shopProduct =
    shopProducts.find(
      (item) =>
        item.slug === slug
    );

  /*
   * BASIC PRODUCT OPTIONS
   */

  const [mode, setMode] =
    useState<
      'quick' | 'custom'
    >('quick');

  const [
    designSource,
    setDesignSource,
  ] = useState<
    'ours' | 'own'
  >('ours');

  const [
    occasion,
    setOccasion,
  ] = useState('');

  const [
    size,
    setSize,
  ] = useState(
    product?.sizes?.[0] || ''
  );

  const [
    orientation,
    setOrientation,
  ] = useState(
    product?.orientations?.[0] ||
      'portrait'
  );

  const [
    template,
    setTemplate,
  ] = useState(
    product?.templates?.[0] || ''
  );

  const [
    textLayout,
    setTextLayout,
  ] = useState<TextLayout>(
    product?.textLayouts?.[0] ||
      'photo-only'
  );

  const [
    topText,
    setTopText,
  ] = useState('');

  const [
    bottomText,
    setBottomText,
  ] = useState('');

  const [
    paragraph,
    setParagraph,
  ] = useState('');

  const [
    accessory,
    setAccessory,
  ] = useState(
    product?.accessories?.[0] ||
      ''
  );

  const [
    recipient,
    setRecipient,
  ] = useState('');

  const [
    sender,
    setSender,
  ] = useState('');

  /*
   * PHOTO / ARTWORK
   */

  const [
    photo,
    setPhoto,
  ] = useState<File | null>(
    null
  );

  const [
    photoUrl,
    setPhotoUrl,
  ] = useState('');

  /*
   * AUDIO
   */

  const [
    audio,
    setAudio,
  ] = useState<Blob | null>(
    null
  );

  const [
    audioUrl,
    setAudioUrl,
  ] = useState('');

  const [
    recording,
    setRecording,
  ] = useState(false);

  const [
    recordingSeconds,
    setRecordingSeconds,
  ] = useState(0);

  const mediaRecorder =
    useRef<
      MediaRecorder | null
    >(null);

  const audioChunks =
    useRef<Blob[]>([]);

  const recordingTimer =
    useRef<
      ReturnType<
        typeof setInterval
      > | null
    >(null);

  /*
   * SAVING
   */

  const [
    saving,
    setSaving,
  ] = useState(false);

  const [
    savedCode,
    setSavedCode,
  ] = useState<
    string | null
  >(null);

  const [
    savedPhotoUrl,
    setSavedPhotoUrl,
  ] = useState<
    string | null
  >(null);

  const [
    savedAudioUrl,
    setSavedAudioUrl,
  ] = useState<
    string | null
  >(null);

  const [
    finalReview,
    setFinalReview,
  ] = useState(false);

  /*
   * HELPERS
   */

  function markChanged() {
    setFinalReview(false);
  }

  /*
   * PHOTO / DESIGN UPLOAD
   */

  function choosePhoto(
    event:
      React.ChangeEvent<HTMLInputElement>
  ) {
    const file =
      event.target.files?.[0];

    if (!file) {
      return;
    }

    if (photoUrl) {
      URL.revokeObjectURL(
        photoUrl
      );
    }

    setPhoto(file);

    setPhotoUrl(
      URL.createObjectURL(file)
    );

    markChanged();
  }

  /*
   * START RECORDING
   */

  async function startRecording() {
    try {
      const stream =
        await navigator.mediaDevices.getUserMedia(
          {
            audio: true,
          }
        );

      audioChunks.current = [];

      const supportedTypes = [
        'audio/webm;codecs=opus',
        'audio/webm',
        'audio/mp4',
      ];

      const mimeType =
        supportedTypes.find(
          (type) =>
            MediaRecorder.isTypeSupported(
              type
            )
        );

      const recorder =
        new MediaRecorder(
          stream,
          mimeType
            ? {
                mimeType,
              }
            : undefined
        );

      mediaRecorder.current =
        recorder;

      recorder.ondataavailable = (
        event
      ) => {
        if (
          event.data.size > 0
        ) {
          audioChunks.current.push(
            event.data
          );
        }
      };

      recorder.onstop = () => {
        const type =
          recorder.mimeType ||
          audioChunks.current[0]
            ?.type ||
          'audio/webm';

        const blob =
          new Blob(
            audioChunks.current,
            {
              type,
            }
          );

        if (audioUrl) {
          URL.revokeObjectURL(
            audioUrl
          );
        }

        setAudio(blob);

        setAudioUrl(
          URL.createObjectURL(
            blob
          )
        );

        stream
          .getTracks()
          .forEach(
            (track) =>
              track.stop()
          );

        markChanged();
      };

      recorder.start(250);

      setRecordingSeconds(0);

      setRecording(true);

      recordingTimer.current =
        setInterval(() => {
          setRecordingSeconds(
            (seconds) =>
              seconds + 1
          );
        }, 1000);
    } catch {
      alert(
        'Microphone access failed. Please allow microphone permission or upload an audio message instead.'
      );
    }
  }

  /*
   * STOP RECORDING
   */

  function stopRecording() {
    if (
      mediaRecorder.current &&
      mediaRecorder.current
        .state !== 'inactive'
    ) {
      mediaRecorder.current.stop();
    }

    if (
      recordingTimer.current
    ) {
      clearInterval(
        recordingTimer.current
      );

      recordingTimer.current =
        null;
    }

    setRecording(false);
  }

  /*
   * DELETE RECORDING
   */

  function deleteRecording() {
    if (audioUrl) {
      URL.revokeObjectURL(
        audioUrl
      );
    }

    setAudio(null);
    setAudioUrl('');
    setRecordingSeconds(0);

    markChanged();
  }

  /*
   * UPLOAD AUDIO
   */

  function chooseAudio(
    event:
      React.ChangeEvent<HTMLInputElement>
  ) {
    const file =
      event.target.files?.[0];

    if (!file) {
      return;
    }

    if (audioUrl) {
      URL.revokeObjectURL(
        audioUrl
      );
    }

    setAudio(file);

    setAudioUrl(
      URL.createObjectURL(file)
    );

    setRecordingSeconds(0);

    markChanged();
  }

  /*
   * SAVE SMART MEMORY
   */

  async function saveSmartMemory() {
    if (!recipient.trim()) {
      alert(
        'Please enter who this gift is for.'
      );

      return;
    }

    if (!photo) {
      alert(
        product?.slug ===
          'photo-mug' ||
        product?.slug ===
          'coaster-set'
          ? 'Please upload your photograph or design.'
          : 'Please add a photograph before continuing.'
      );

      return;
    }

    if (recording) {
      alert(
        'Please stop the voice recording first.'
      );

      return;
    }

    setSaving(true);

    try {
      const code =
        makeCode();

      const formData =
        new FormData();

      formData.set(
        'code',
        code
      );

      formData.set(
        'recipient',
        recipient.trim()
      );

      const memoryMessage =
        paragraph.trim() ||
        bottomText.trim() ||
        topText.trim() ||
        occasion ||
        'Smart Memory';

      formData.set(
        'message',
        memoryMessage
      );

      formData.set(
        'sender',
        sender.trim()
      );

      formData.set(
        'photo',
        photo
      );

      if (audio) {
        const extension =
          audio.type.includes(
            'mp4'
          )
            ? 'm4a'
            : 'webm';

        formData.set(
          'audio',
          new File(
            [audio],
            `message.${extension}`,
            {
              type:
                audio.type ||
                'audio/webm',
            }
          )
        );
      }

      const response =
        await fetch(
          '/api/memories',
          {
            method: 'POST',
            body: formData,
          }
        );

      const result =
        await response.json();

      if (!response.ok) {
        throw new Error(
          result.error ||
            'Could not save Smart Memory'
        );
      }

      setSavedCode(
        result.code ||
          result.memory?.code ||
          code
      );

      setSavedPhotoUrl(
        result.photoUrl ||
          result.memory
            ?.photoUrl ||
          result.memory
            ?.photo_url ||
          null
      );

      setSavedAudioUrl(
        result.audioUrl ||
          result.memory
            ?.audioUrl ||
          result.memory
            ?.audio_url ||
          null
      );

      setFinalReview(true);
    } catch (error) {
      alert(
        error instanceof Error
          ? error.message
          : 'Could not save Smart Memory'
      );
    } finally {
      setSaving(false);
    }
  }

  /*
   * ADD TO BASKET
   */

  function addToBasket() {
    if (
      !product ||
      !savedCode ||
      !finalReview
    ) {
      return;
    }

    const price =
      shopProduct?.price ??
      0;

    const basketItem = {
      id:
        crypto.randomUUID(),

      productType:
        'product',

      productSlug:
        product.slug,

      memoryCode:
        savedCode,

      name:
        product.name,

      category:
        occasion ||
        shopProduct?.category ||
        'Personalised Gift',

      recipient:
        recipient.trim(),

      imageUrl:
        savedPhotoUrl ||
        photoUrl ||
        null,

      audioUrl:
        savedAudioUrl,

      price,

      quantity: 1,

      personalisation: {
        mode,

        designSource,

        occasion,

        size,

        orientation,

        template:
          designSource ===
            'ours' &&
          mode === 'quick'
            ? template
            : null,

        textLayout,

        topText,

        bottomText,

        paragraph,

        accessory,

        sender,

        coasterQuantity:
          product.slug ===
          'coaster-set'
            ? size
            : null,
      },
    };

    try {
      const existing =
        localStorage.getItem(
          'smart-memories-cart'
        );

      const parsed =
        existing
          ? JSON.parse(
              existing
            )
          : [];

      const basket =
        Array.isArray(parsed)
          ? parsed
          : [];

      basket.push(
        basketItem
      );

      localStorage.setItem(
        'smart-memories-cart',
        JSON.stringify(
          basket
        )
      );

      window.location.href =
        '/cart';
    } catch {
      alert(
        'Your Smart Memory was saved, but the product could not be added to the basket.'
      );
    }
  }

  /*
   * PRODUCT CHECK
   */

  if (!product) {
    return (
      <section className="wrap">
        <h1>
          Product not found
        </h1>

'use client';

import { useEffect, useRef, useState } from 'react';
import Link from 'next/link';

type CloudMemory = {
  code: string;
  recipient: string;
  message: string;
  sender: string;
  photoUrl?: string | null;
  audioUrl?: string | null;
};

export default function Memory({
  params,
}: {
  params: Promise<{ code: string }>;
}) {
  const [m, setM] = useState<CloudMemory | null | undefined>(undefined);
  const [playing, setPlaying] = useState(false);
  const [audioReady, setAudioReady] = useState(false);
  const [audioError, setAudioError] = useState(false);

  const player = useRef<HTMLAudioElement>(null);

  useEffect(() => {
    params.then(async ({ code }) => {
      try {
        const r = await fetch(
          '/api/memories/' + encodeURIComponent(code),
          { cache: 'no-store' }
        );

        if (!r.ok) {
          setM(null);
          return;
        }

        setM(await r.json());
      } catch {
        setM(null);
      }
    });
  }, [params]);

  async function play() {
    const audio = player.current;

    if (!audio || !audioReady) return;

    try {
      if (audio.paused) {
        await audio.play();
      } else {
        audio.pause();
      }
    } catch {
      setAudioError(true);
    }
  }

  if (m === undefined) {
    return (
      <section className="wrap">
        <p>Loading memory...</p>
      </section>
    );
  }

  if (!m) {
    return (
      <section className="wrap">
        <div className="card">
          <h1>Memory not found</h1>
          <p>This memory could not be found online.</p>
          <Link className="btn" href="/">
            Smart Memories
          </Link>
        </div>
      </section>
    );
  }

  return (
    <section className="memory">
      <span className="pill">MEMORY FOR</span>

      <h1>{m.recipient}</h1>

      {m.photoUrl && (
        <img
          className="memoryphoto"
          src={m.photoUrl}
          alt={m.recipient}
        />
      )}

      <div className="card">
        <p className="quote">
          {m.message || 'A memory made especially for you.'}
        </p>

        {m.audioUrl ? (
          <>
            <audio
              ref={player}
              src={m.audioUrl}
              preload="auto"
              onCanPlay={() => setAudioReady(true)}
              onCanPlayThrough={() => setAudioReady(true)}
              onLoadedData={() => setAudioReady(true)}
              onWaiting={() => setAudioReady(false)}
              onEnded={() => setPlaying(false)}
              onPause={() => setPlaying(false)}
              onPlay={() => {
                setPlaying(true);
                setAudioReady(true);
              }}
              onError={() => {
                setAudioError(true);
                setAudioReady(false);
              }}
            />

            {!audioReady && !audioError && (
              <p className="success">
                Preparing your voice message...
              </p>
            )}

            <button
              className="btn big play"
              onClick={play}
              disabled={!audioReady || audioError}
            >
              {audioError
                ? 'VOICE MESSAGE UNAVAILABLE'
                : !audioReady
                  ? 'LOADING VOICE MESSAGE...'
                  : playing
                    ? '⏸ PAUSE MY MESSAGE'
                    : '▶ PLAY MY MESSAGE'}
            </button>

            {!audioError && audioReady && (
              <p className="success">
                🎙 Voice message ready
              </p>
            )}
          </>
        ) : (
          <p>No voice message was added to this memory.</p>
        )}

        {m.sender && (
          <p>
            With love always, <b>{m.sender}</b> ♡
          </p>
        )}

        <small>Memory code: {m.code}</small>
      </div>
    </section>
  );
}

import { NextResponse } from 'next/server';
import { randomBytes } from 'crypto';
import { supabaseAdmin } from '@/lib/supabaseAdmin';

export const runtime = 'nodejs';

/* =========================================================
   FILE EXTENSION
   ========================================================= */

function ext(
  name: string,
  fallback: string
) {
  const extension =
    name
      .split('.')
      .pop()
      ?.toLowerCase();

  return extension &&
    /^[a-z0-9]+$/.test(
      extension
    )
    ? extension
    : fallback;
}

/* =========================================================
   SMART MEMORY CODE
   Example: V5JYH4L3
   ========================================================= */

function generateMemoryCode() {
  const characters =
    'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';

  const bytes =
    randomBytes(8);

  let code = '';

  for (
    let i = 0;
    i < 8;
    i++
  ) {
    code +=
      characters[
        bytes[i] %
          characters.length
      ];
  }

  return code;
}

export async function POST(

      )
        .trim()
        .toUpperCase();

    const code =

      )
        ? suppliedCode
        : generateMemoryCode();

    /* -----------------------------------------------------
       CUSTOMER DATA
       ----------------------------------------------------- */

    const recipient =
      String(
        form.get(
          'recipient'
        ) || ''
      ).trim();

    const message =
      String(
        form.get(
          'message'
        ) || ''
      ).trim();

    const sender =
      String(
        form.get(
          'sender'
        ) || ''
      ).trim();

    if (!recipient) {
      return NextResponse.json(
        {
          error:
            'Please enter the recipient name.',
        },
        {
          status: 400,
        }
      );
    }

    const sb =

    let photoUrl:
      | string
      | null = null;

    let audioUrl:
      | string
      | null = null;

    /* =====================================================
       PHOTO
       ===================================================== */

    const photo =
      form.get('photo');

    if (
      photo instanceof
        File &&
      photo.size >
        0
    ) {
      if (

        return NextResponse.json(
          {
            error:

          },
          {
            status: 400,
          }
        );
      }

      const photoPath =

        )}`;

      } =
        await sb.storage
          .from(
            'memory-media'
          )
          .upload(
            photoPath,

            {
              contentType:
                photo.type ||
                'image/jpeg',

              upsert: true,
            }
          );

      if (
        photoUploadError
      ) {
        throw photoUploadError;
      }

      const {
        data:
          publicPhoto,
      } =
        sb.storage
          .from(
            'memory-media'
          )
          .getPublicUrl(
            photoPath
          );

      photoUrl =
        publicPhoto.publicUrl;
    }

    /* =====================================================
       AUDIO
       ===================================================== */

    const audio =
      form.get('audio');

    if (
      audio instanceof
        File &&
      audio.size >
        0
    ) {
      if (
        audio.size >
        25 *
          1024 *
          1024
      ) {
        return NextResponse.json(
          {
            error:
              'Audio is too large. Maximum size is 25 MB.',
          },
          {
            status: 400,
          }
        );
      }

      const fallbackAudioExtension =
        audio.type.includes(
          'mp4'
        )
          ? 'm4a'
          : 'webm';

      const audioPath =
        `${code}/audio.${ext(
          audio.name,
          fallbackAudioExtension
        )}`;

      const {
        error:
          audioUploadError,
      } =
        await sb.storage
          .from(
            'memory-media'
          )
          .upload(
            audioPath,

      }

      const {
        data:

        sb.storage
          .from(
            'memory-media'
          )
          .getPublicUrl(
            audioPath
          );

      audioUrl =
        publicAudio.publicUrl;
    }

    /* =====================================================
       SAVE TO DATABASE
       ===================================================== */

    );

  function getSupportedAudioMimeType() {
    if (typeof MediaRecorder === 'undefined') {

  async function startRecording() {
    setAudioError(null);
    setSaveSuccess(null);

    if (!navigator.mediaDevices?.getUserMedia) {
      setAudioError(
        'Voice recording is not supported by this browser. Please upload an audio file instead.'
      );
      return;
    }

    try {
      if (recordedAudioUrl) {

      if (uploadedAudioUrl) {
        URL.revokeObjectURL(uploadedAudioUrl);

      setUploadedAudioUrl(null);
      setUploadedAudioFile(null);

      const stream =
        await navigator.mediaDevices.getUserMedia({
          audio: true,
        });

      mediaStreamRef.current = stream;

      const mimeType = getSupportedAudioMimeType();

      const recorder = mimeType
        ? new MediaRecorder(stream, { mimeType })
        : new MediaRecorder(stream);

      mediaRecorderRef.current = recorder;
      audioChunksRef.current = [];

      recorder.ondataavailable = (event) => {
        if (event.data.size > 0) {
          audioChunksRef.current.push(event.data);
        }
      };

      recorder.onstop = () => {
        const finalType =
          recorder.mimeType ||
          audioChunksRef.current[0]?.type ||
          'audio/webm';

        const audioBlob = new Blob(
          audioChunksRef.current,
          {
            type: finalType,
          }
        );

        const audioUrl =
          URL.createObjectURL(audioBlob);

        setRecordedAudioBlob(audioBlob);
        setRecordedAudioUrl(audioUrl);

        if (mediaStreamRef.current) {
          mediaStreamRef.current
            .getTracks()
            .forEach((track) => track.stop());
        }

        mediaStreamRef.current = null;
      };

      recorder.start();

      setRecordingSeconds(0);
      setIsRecording(true);

      timerRef.current = setInterval(() => {
        setRecordingSeconds((seconds) => seconds + 1);
      }, 1000);
    } catch (error) {
      console.error(error);

      setAudioError(
        'Microphone access was not available. Please allow microphone permission in your browser and try again.'
      );
    }
  }

  function stopRecording() {
    if (
      mediaRecorderRef.current &&
      mediaRecorderRef.current.state !== 'inactive'
    ) {
      mediaRecorderRef.current.stop();
    }

    setIsRecording(false);

    if (timerRef.current) {
      clearInterval(timerRef.current);
      timerRef.current = null;
    }
  }

  function deleteRecording() {
    if (recordedAudioUrl) {

    setRecordedAudioUrl(null);
    setRecordedAudioBlob(null);
    setRecordingSeconds(0);
    setAudioError(null);
  }

  function addSavedCardToBasket() {
    if (!savedPersonalisationId) {
      setSaveError('Please save the personalised card first.');
      return;
    }

    const basketItem = {
      id: crypto.randomUUID(),
      personalisationId: savedPersonalisationId,

        cardType === 'smart'
          ? 'smart-card'
          : 'traditional-card',

      name:
        cardType === 'smart'
          ? `${occasion} Smart Memory Card`
          : `${occasion} Handcrafted Card`,

      occasion,
      design,
      recipient,

      imageUrl: savedMemoryPhotoUrl,

      price: Number(price),
      quantity: 1,
    };

    try {
      const existingBasket =
        localStorage.getItem('smart-memories-cart');

      const parsedBasket = existingBasket
        ? JSON.parse(existingBasket)
        : [];

      const basket = Array.isArray(parsedBasket)
        ? parsedBasket
        : [];

      basket.push(basketItem);

      localStorage.setItem(
        'smart-memories-cart',
        JSON.stringify(basket)
      );

      router.push('/cart');
    } catch (error) {
      console.error(error);

      setSaveError(
        'Your card was saved, but it could not be added to the basket.'
      );
    }

  async function savePersonalisation() {

      return;

    try {
      setIsSaving(true);

      const formData = new FormData();

      formData.append('occasion', occasion);
      formData.append('designName', design);
      formData.append('cardType', cardType);
      formData.append('recipientName', recipient);
      formData.append('frontGreeting', frontGreeting);
      formData.append('insideMessage', insideMessage);
      formData.append('senderName', sender);
      formData.append('colour', colour);

      formData.append(
        'includePrintedPhoto',
        String(includePhoto)
      );

      formData.append('totalPrice', price);

      if (
        cardType === 'smart' &&
        memoryPhotoFile
      ) {
        formData.append(
          'memoryPhoto',
          memoryPhotoFile
        );
      }

      if (
        cardType === 'smart' &&
        recordedAudioBlob
      ) {
        const extension =
          getAudioFileExtension(
            recordedAudioBlob.type
          );

        const voiceFile = new File(
          [recordedAudioBlob],
          `voice-message.${extension}`,
          {
            type:
              recordedAudioBlob.type ||
              'audio/webm',
          }
        );

        formData.append(
          'audioFile',
          voiceFile
        );
      } else if (
        cardType === 'smart' &&
        uploadedAudioFile
      ) {
        formData.append(
          'audioFile',
          uploadedAudioFile
        );
      }

      const response = await fetch(
        '/api/card-personalisations',
        {
          method: 'POST',
          body: formData,
        }
      );

      const result = await response.json();

      if (!response.ok) {
        throw new Error(
          result.error ||
            'Could not save your personalised card.'
        );
      }

      setSavedPersonalisationId(
        result.personalisation.id
      );

      setSavedMemoryPhotoUrl(
        result.personalisation.memoryPhotoUrl || null
      );

      setSaveSuccess(
        'Your personalised card and Smart Memory have been saved successfully.'
      );
    } catch (error) {
      console.error(error);

      setSaveError(
        error instanceof Error
          ? error.message
          : 'Something went wrong while saving your card.'
      );
    } finally {
      setIsSaving(false);
    }
